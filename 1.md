import json
from pathlib import Path

root = Path("/home/ubuntu/Extraction-Reviewer")
extracted = root / "extracted"
assignment_path = root / "data/users/Xuben/assignments.json"

# 只识别完整的论文目录
paper_ids = sorted(
    p.name for p in extracted.iterdir()
    if p.is_dir()
    and (p / "extraction_postprocess" / "field_evidence.json").is_file()
)

data = json.loads(assignment_path.read_text(encoding="utf-8"))
old_papers = data.get("papers", [])

# 保留已有分配，只追加，不重复
data["papers"] = list(dict.fromkeys(old_papers + paper_ids))

assignment_path.write_text(
    json.dumps(data, ensure_ascii=False, indent=2) + "\n",
    encoding="utf-8",
)

print(f"原有论文数: {len(old_papers)}")
print(f"extracted 中有效论文数: {len(paper_ids)}")
print(f"现在分配论文数: {len(data['papers'])}")
PY
